import logging

import requests
from fastapi import APIRouter
from fastapi import HTTPException
from app.schemas import VerifyOTPRequest
from app.schemas import GenerateOTPRequest
from app.redis_client import redis_client
from app.otp import generate_otp
from app.security import hash_otp
from app.services.whatsapp_service import send_otp as send_whatsapp_otp
import os

logger = logging.getLogger(__name__)

MAX_OTP_ATTEMPTS = int(os.getenv("MAX_OTP_ATTEMPTS", 5))
OTP_LOCK_TIME = int(os.getenv("OTP_LOCK_TIME", 900))

router = APIRouter(
    prefix="/api/auth",
    tags=["Authentication"]
)


OTP_EXPIRY = int(os.getenv("OTP_EXPIRY", 300))

@router.post("/generate-otp")
async def generate_otp_api(data: GenerateOTPRequest):

    cooldown_key = f"cooldown:{data.phone}"

    if redis_client.exists(cooldown_key):
        raise HTTPException(
            status_code=429,
            detail="Please wait before requesting another OTP."
        )

    otp = generate_otp()

    hashed_otp = hash_otp(otp)

    redis_client.setex(
        f"otp:{data.phone}",
        OTP_EXPIRY,
        hashed_otp
)

    redis_client.setex(
        cooldown_key,
        30,
        "locked"
    )

    try:
        send_whatsapp_otp(data.phone, otp)
    except requests.exceptions.HTTPError as e:
        redis_client.delete(f"otp:{data.phone}")
        redis_client.delete(cooldown_key)

        provider_message = "Failed to send OTP via WhatsApp"
        if e.response is not None:
            try:
                provider_message = e.response.json().get("message", provider_message)
            except ValueError:
                pass

        logger.error(
            "WhatsApp OTP send failed for %s: %s",
            data.phone,
            e.response.text if e.response is not None else str(e)
        )
        raise HTTPException(
            status_code=502,
            detail=f"Failed to send OTP via WhatsApp: {provider_message}"
        )
    except Exception:
        redis_client.delete(f"otp:{data.phone}")
        redis_client.delete(cooldown_key)
        logger.exception("WhatsApp OTP send failed for %s", data.phone)
        raise HTTPException(
            status_code=502,
            detail="Failed to send OTP via WhatsApp"
        )

    return {
        "status": "success",
        "message": "OTP sent via WhatsApp",
        "phone": data.phone
    }


@router.post("/verify-otp")
async def verify_otp(data: VerifyOTPRequest):

    otp_key = f"otp:{data.phone}"
    attempt_key = f"attempt:{data.phone}"
    lock_key = f"lock:{data.phone}"

    # Check if phone is locked
    if redis_client.exists(lock_key):
        raise HTTPException(
            status_code=429,
            detail="Too many failed attempts. Try again later."
        )

    stored_otp = redis_client.get(otp_key)

    if stored_otp is None:
        raise HTTPException(
            status_code=404,
            detail="OTP expired or not found"
        )

    # OTP matches
    if stored_otp == hash_otp(data.otp):

        redis_client.delete(otp_key)
        redis_client.delete(attempt_key)

        return {
            "status": "success",
            "message": "OTP Verified Successfully"
        }

    # OTP is incorrect
    attempts = redis_client.incr(attempt_key)

    # Set expiry on first failed attempt
    if attempts == 1:
        redis_client.expire(attempt_key, 300)

    remaining = MAX_OTP_ATTEMPTS - attempts

    if attempts >= MAX_OTP_ATTEMPTS:

        redis_client.delete(otp_key)
        redis_client.delete(attempt_key)

        redis_client.setex(
            lock_key,
            OTP_LOCK_TIME,
            "locked"
        )

        raise HTTPException(
            status_code=429,
            detail="Maximum attempts exceeded. Phone number temporarily locked."
        )

    raise HTTPException(
        status_code=400,
        detail=f"Invalid OTP. {remaining} attempts remaining."
    )
